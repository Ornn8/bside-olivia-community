Set shell = CreateObject("WScript.Shell")
Set fso = CreateObject("Scripting.FileSystemObject")
root = fso.GetParentFolderName(WScript.ScriptFullName)
' The interface language saved by the app (data\config\ui-language.json); Chinese unless English was chosen.
Function OliviaEnglish(installRoot)
    OliviaEnglish = False
    On Error Resume Next
    savedLanguage = installRoot & "\data\config\ui-language.json"
    If fso.FileExists(savedLanguage) Then
        Set languageStream = fso.OpenTextFile(savedLanguage, 1)
        languageText = languageStream.ReadAll
        languageStream.Close
        If InStr(languageText, """language"": ""en""") > 0 Then OliviaEnglish = True
    End If
    On Error GoTo 0
End Function
If WScript.Arguments.Named.Exists("error") Then
    errorCode = WScript.Arguments.Named.Item("error")
    failureTitle = "Olivia 本地版"
    failureText = "Olivia 启动失败（错误码 " & errorCode & "）。" & vbCrLf & "请将安装目录 data\logs 下的启动日志提供给维护者。"
    If OliviaEnglish(root) Then
        failureTitle = "Olivia"
        failureText = "Olivia failed to start (error code " & errorCode & ")." & vbCrLf & "Please send the startup logs in the data\logs folder of the installation to the maintainer."
    End If
    MsgBox failureText, 16, failureTitle
    WScript.Quit 0
End If
wscriptExe = shell.ExpandEnvironmentStrings("%WINDIR%") & "\System32\wscript.exe"
startCommand = root & "\START.cmd"
launchErrorNumber = 0
launchErrorDescription = ""
If Not fso.FileExists(startCommand) Then
    launchErrorNumber = 2
    launchErrorDescription = "START_CMD_NOT_FOUND"
Else
    On Error Resume Next
    exitCode = shell.Run(Chr(34) & startCommand & Chr(34), 0, True)
    launchErrorNumber = Err.Number
    launchErrorDescription = Err.Description
    Err.Clear
    On Error GoTo 0
End If
If launchErrorNumber <> 0 Then
    exitCode = 2
End If
If exitCode <> 0 Then
    If exitCode <> -1 And CStr(exitCode) <> "4294967295" Then
    ' Batch mode also suppresses COM popups: explicitly use an interactive error host.
    errorCommand = Chr(34) & wscriptExe & Chr(34) & " //I //Nologo " & Chr(34) & WScript.ScriptFullName & Chr(34) & " /error:" & CStr(exitCode)
    shell.Run errorCommand, 0, True
End If
End If
WScript.Quit exitCode
